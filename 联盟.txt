local dragCard, dsC, csC = false, nil, nil
title.InputBegan:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        dragCard = true
        dsC = i.Position
        csC = card.Position
    end
end)
UIS.InputChanged:Connect(function(i)
    if not dragCard then return end
    if i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch then
        local d = i.Position - dsC
        card.Position = UDim2.new(csC.X.Scale, csC.X.Offset + d.X, csC.Y.Scale, csC.Y.Offset + d.Y)
    end
end)
title.InputEnded:Connect(function() dragCard = false end)

local dragBtn, dsB, csB, movedBtn = false, nil, nil, false
local CLICK_THRESHOLD = 15

toggleBtn.InputBegan:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        dragBtn = true
        movedBtn = false
        dsB = i.Position
        csB = toggleBtn.Position
    end
end)

UIS.InputChanged:Connect(function(i)
    if not dragBtn then return end
    if i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch then
        local d = i.Position - dsB
        if d.Magnitude > CLICK_THRESHOLD then
            movedBtn = true
            toggleBtn.Position = UDim2.new(csB.X.Scale, csB.X.Offset + d.X, csB.Y.Scale, csB.Y.Offset + d.Y)
        end
    end
end)

toggleBtn.InputEnded:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        if not movedBtn then
            card.Visible = not card.Visible
        end
        dragBtn = false
        movedBtn = false
    end
end)

local FE
task.spawn(function()
    FE = RS:WaitForChild("Fishing"):WaitForChild("Remotes"):WaitForChild("FishingEvent")
end)

local function getAFKRemote()
    local RE = RS:FindFirstChild("RemoteEvents")
    return RE and RE:FindFirstChild("AFKRemote")
end

local function getTF()
    local g = P.PlayerGui:FindFirstChild("_LobbyUI")
    local f = g and g:FindFirstChild("UPDATE: 10 UI Folder")
    local fc = f and f:FindFirstChild("Fish Content")
    local fish = fc and fc:FindFirstChild("Fishing")
    return fish and fish:FindFirstChild("TargetFrame")
end

local function nowTime() return tick() end

local running = false
local currentIndex = 0
local lastHit = 0
local roundId = 0
local totalCaught = 0
local afkOn = false

local function updateStat()
    statLine.Text = string.format("蓄力 %.1fs   已钓 %d", HOLD_TIME, totalCaught)
end

local function setUI(state)
    if state == "running" then
        dot.BackgroundColor3 = Color3.fromRGB(80, 220, 120)
        statusText.Text = "运行中"
        statusText.TextColor3 = Color3.fromRGB(120, 220, 150)
        btn.Text = "停止"
        btnGrad.Color = ColorSequence.new{
            ColorSequenceKeypoint.new(0, Color3.fromRGB(210, 90, 90)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(160, 55, 55)),
        }
    else
        dot.BackgroundColor3 = Color3.fromRGB(100, 100, 120)
        statusText.Text = "已停止"
        statusText.TextColor3 = Color3.fromRGB(140, 150, 170)
        btn.Text = "开始"
        btnGrad.Color = ColorSequence.new{
            ColorSequenceKeypoint.new(0, Color3.fromRGB(85, 130, 200)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(55, 85, 140)),
        }
    end
end

local function startRound()
    if not running then return end
    if not FE then return end

    currentIndex = 0
    roundId = roundId + 1
    local myId = roundId

    infoLine.Text = "抛竿中..."
    -- ★ 使用固定钓鱼坐标
    FE:FireServer("Cast", {Position = CAST_POS})

    task.delay(30, function()
        if roundId == myId and running then
            startRound()
        end
    end)
end

local function startFishing()
    if running then return end
    running = true
    roundId = roundId + 1
    currentIndex = 0
    setUI("running")
    task.spawn(function()
        if not FE then task.wait(0.5) end
        task.wait(0.3)
        startRound()
    end)
end

local function stopFishing()
    running = false
    roundId = roundId + 1
    currentIndex = 0
    lastHit = 0
    setUI("stopped")
    infoLine.Text = ""
end

local function doTeleport()
    local c = P.Character
    if not c then footLine.Text = "角色不存在" return false end
    local h = c:FindFirstChild("HumanoidRootPart")
    if not h then footLine.Text = "找不到 HRP" return false end
    h.CFrame = CFrame.new(TELEPORT_POS)   -- ★ 用新坐标
    footLine.Text = "已传送"
    return true
end

local function enableAFK()
    local afk = getAFKRemote()
    if not afk then footLine.Text = "找不到 AFKRemote" return end
    afk:FireServer("Start")
    afkOn = true
    afkBtn.Text = "关闭 AFK"
    afkGrad.Color = ColorSequence.new{
        ColorSequenceKeypoint.new(0, Color3.fromRGB(80, 190, 120)),
        ColorSequenceKeypoint.new(1, Color3.fromRGB(55, 140, 85)),
    }
    footLine.Text = "AFK 已开启"
    print("[AFK] Start")
end

local function disableAFK()
    local afk = getAFKRemote()
    if afk then
        for _, act in ipairs({"Stop", "End", "Cancel", "Close", "Leave", "Idle", "StopAFK"}) do
            pcall(function() afk:FireServer(act) end)
        end
    end
    afkOn = false
    afkBtn.Text = "开启 AFK"
    afkGrad.Color = ColorSequence.new{
        ColorSequenceKeypoint.new(0, Color3.fromRGB(70, 110, 170)),
        ColorSequenceKeypoint.new(1, Color3.fromRGB(45, 75, 120)),
    }
    footLine.Text = "AFK 已关闭"
    print("[AFK] Stop")
end

task.spawn(function()
    while not FE do task.wait(0.2) end

    FE.OnClientEvent:Connect(function(action, data)
        if not running then return end
        if type(data) ~= "table" then return end

        if action == "CastStarted" then
            infoLine.Text = "蓄力中..."
            task.spawn(function()
                task.wait(0.3)
                if not running then return end
                FE:FireServer("LuckHold", {ClickTime = nowTime()})
                local endT = tick() + HOLD_TIME
                while tick() < endT do
                    if not running then return end
                    FE:FireServer("LuckHold", {ClickTime = nowTime()})
                    task.wait(0.1)
                end
                if not running then return end
                infoLine.Text = "释放..."
                FE:FireServer("LuckRelease", {ClickTime = nowTime()})
            end)

        elseif action == "LuckLocked" then
            local luck = data.Luck or 0
            infoLine.Text = string.format("x%.2f", luck)

        elseif action == "Target" then
            currentIndex = data.Index or 0
            infoLine.Text = "判定 " .. currentIndex

        elseif action == "Catch" then
            totalCaught = totalCaught + 1
            updateStat()
            infoLine.Text = "钓到!"
            task.wait(0.3)
            if running then startRound() end
        end
    end)
end)

task.spawn(function()
    while true do
        task.wait(0.03)
        if running and currentIndex > 0 then
            local tf = getTF()
            if tf and tf.Visible then
                local cl = tf:FindFirstChild("Click")
                if cl and cl.Visible and cl.AbsoluteSize.X > 0 then
                    local now = tick()
                    if now - lastHit >= HIT_GAP then
                        lastHit = now
                        FE:FireServer("Hit", {
                            Index = currentIndex,
                            ClickTime = nowTime()
                        })
                    end
                end
            end
        end
    end
end)

btn.Activated:Connect(function()
    if running then stopFishing() else startFishing() end
end)

afkBtn.Activated:Connect(function()
    if afkOn then disableAFK() else enableAFK() end
end)

tpBtn.Activated:Connect(function()
    doTeleport()
end)

-- 自启动
task.spawn(function()
    if not P.Character then P.CharacterAdded:Wait() end
    task.wait(1)

    for i = 1, 5 do
        local h = P.Character and P.Character:FindFirstChild("HumanoidRootPart")
        if h then
            h.CFrame = CFrame.new(TELEPORT_POS)   -- ★ 用新坐标
            footLine.Text = "启动传送完成"
            break
        end
        task.wait(0.5)
    end

    task.wait(0.3)
    enableAFK()

    if AUTO_START then
        task.wait(0.5)
        startFishing()
    end

    print("[自启动] 传送 + AFK + 钓鱼 已全部启动")
end)
