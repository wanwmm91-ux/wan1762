local OPTION_DEFAULT_BG = Color3.fromRGB(40, 50, 70)
local OPTION_SELECTED_BG = Color3.fromRGB(70, 150, 200)
local OPTION_DEFAULT_TC = Color3.fromRGB(200, 215, 235)
local OPTION_SELECTED_TC = Color3.fromRGB(255, 255, 255)

local selectedAction = nil
local allOptionBtns = {opt10M, optSci, optParty, optFishC, optLuck, optSummon, optSpin}

local function clearSelection()
    selectedAction = nil
    for _, ob in ipairs(allOptionBtns) do
        ob.BackgroundColor3 = OPTION_DEFAULT_BG
        ob.TextColor3 = OPTION_DEFAULT_TC
    end
    drawStatus.Text = "未选择"
    drawStatus.TextColor3 = Color3.fromRGB(140, 150, 170)
end

local function selectOption(btn, key)
    selectedAction = key
    for _, ob in ipairs(allOptionBtns) do
        if ob == btn then
            ob.BackgroundColor3 = OPTION_SELECTED_BG
            ob.TextColor3 = OPTION_SELECTED_TC
        else
            ob.BackgroundColor3 = OPTION_DEFAULT_BG
            ob.TextColor3 = OPTION_DEFAULT_TC
        end
    end
    drawStatus.Text = "已选: " .. btn.Text
    drawStatus.TextColor3 = Color3.fromRGB(120, 220, 150)
end

local function bindOption(btn)
    btn.Activated:Connect(function()
        local key = btn:GetAttribute("ActionKey")
        if selectedAction == key then
            clearSelection()
        else
            selectOption(btn, key)
        end
    end)
end

for _, ob in ipairs(allOptionBtns) do
    bindOption(ob)
end

    if wasRunning then
        running = false

    lastAutoSwitch = tick()
end

local function isSellFish(name)
    for _, f in ipairs(SELL_FISH) do
        if f == name then return true end
    end
    return false
end

local function scanInventory()
    local sf = getFishScrollFrame()
    if not sf then return {} end

    local list = {}
    for _, child in ipairs(sf:GetChildren()) do
        if child:IsA("Frame")
        and child.Name ~= "Template"
        and child.Name ~= "_dont touch" then

            local move = child:FindFirstChild("Move")
            if move then
                local titleObj = move:FindFirstChild("_Title")
                if titleObj and titleObj:IsA("TextLabel") then
                    local txt = titleObj.Text or ""
                    local w = tonumber(txt:match("([%d%.]+)"))
                    if w then
                        table.insert(list, {
                            name = child.Name,
                            weight = w,
                        })
                    end
                end
            end
        end
    end
    return list
end

local selling = false

local function sellAll()
    if selling then return end
    selling = true

    sellDot.BackgroundColor3 = Color3.fromRGB(255, 200, 80)
    sellStatus.Text = "传送到市场..."
    sellStatus.TextColor3 = Color3.fromRGB(255, 200, 120)
    sellInfo.Text = ""

    if not FF then
        sellStatus.Text = "没找到 FishingFunction"
        sellStatus.TextColor3 = Color3.fromRGB(255, 120, 120)
        selling = false
        return
    end

    task.spawn(function()
        teleportToMarket()
        task.wait(1)

        sellStatus.Text = "扫描中..."
        local items = scanInventory()

        local total = 0
        local counts = {}
        local skipped = 0

        sellStatus.Text = "卖出中..."
        for _, item in ipairs(items) do
            if isSellFish(item.name) then
                pcall(function()
                    FF:InvokeServer("SellFish", {
                        Fish = item.name,
                        Weight = item.weight,
                        Amount = 1
                    })
                end)
                total = total + 1
                counts[item.name] = (counts[item.name] or 0) + 1
                task.wait(0.08)
            else
                skipped = skipped + 1
            end
        end

        if total == 0 then
            sellStatus.Text = "没有可卖的鱼"
            sellStatus.TextColor3 = Color3.fromRGB(255, 200, 120)
            sellInfo.Text = string.format("扫描到 %d 条, 都不在售卖列表", #items)
        else
            sellStatus.Text = "已卖 " .. total .. " 条"
            sellStatus.TextColor3 = Color3.fromRGB(120, 220, 150)
            sellDot.BackgroundColor3 = Color3.fromRGB(80, 220, 120)

            local lines = {}
            for name, cnt in pairs(counts) do
                table.insert(lines, string.format("%s ×%d", name, cnt))
            end
            local show = table.concat(lines, "\n")
            if skipped > 0 then
                show = show .. string.format("\n(跳过 %d 条非列表鱼)", skipped)
            end
            if #show > 250 then show = show:sub(1, 250) .. "\n..." end
            sellInfo.Text = show
        end

        selling = false
    end)
end

local function scanOnly()
    sellStatus.Text = "扫描中..."
    sellStatus.TextColor3 = Color3.fromRGB(140, 180, 220)
    task.spawn(function()
        local items = scanInventory()
        if #items == 0 then
            sellStatus.Text = "没扫到鱼"
            sellStatus.TextColor3 = Color3.fromRGB(255, 200, 120)
            sellInfo.Text = "可能背包没加载"
            return
        end

        local counts = {}
        for _, it in ipairs(items) do
            counts[it.name] = (counts[it.name] or 0) + 1
        end

        local lines = {}
        local sellable = 0
        for name, cnt in pairs(counts) do
            local tag = isSellFish(name) and "✓" or "✗"
            table.insert(lines, string.format("%s %s ×%d", tag, name, cnt))
            if isSellFish(name) then sellable = sellable + cnt end
        end

        sellStatus.Text = string.format("总 %d 条 / 可卖 %d", #items, sellable)
        sellStatus.TextColor3 = Color3.fromRGB(120, 220, 150)
        sellInfo.Text = table.concat(lines, "\n")
    end)
end

    local count = tonumber(countBox.Text)

    if not recipe then
        craftStatus.Text = "配方编号无效"
        craftStatus.TextColor3 = Color3.fromRGB(255, 120, 120)
        return
    end
    if not count or count <= 0 then
        craftStatus.Text = "合成次数无效"
        craftStatus.TextColor3 = Color3.fromRGB(255, 120, 120)
        return
    end

    crafting = true
    craftDot.BackgroundColor3 = Color3.fromRGB(255, 200, 80)
    craftStatus.Text = "传送到市场..."
    craftStatus.TextColor3 = Color3.fromRGB(255, 200, 120)
    craftInfo.Text = ""

    task.spawn(function()
        teleportToMarket()
        task.wait(1)

        local done = 0
        for i = 1, count do
            if not crafting then break end

            pcall(function()
                FF:InvokeServer("Craft", {Recipe = recipe})
            end)

            done = done + 1
            craftStatus.Text = string.format("合成 %d/%d (配方%d)", done, count, recipe)
            craftStatus.TextColor3 = Color3.fromRGB(120, 220, 150)
            craftInfo.Text = string.format("配方: %d  目标: %d", recipe, count)

            task.wait(0.35)
        end

        if done >= count then
            craftStatus.Text = "合成完成 " .. done .. " 次"
            craftStatus.TextColor3 = Color3.fromRGB(120, 220, 150)
            craftDot.BackgroundColor3 = Color3.fromRGB(80, 220, 120)
        else
            craftStatus.Text = "已停止 (完成 " .. done .. ")"
            craftStatus.TextColor3 = Color3.fromRGB(255, 200, 120)
        end

        crafting = false
    end)
end

local function stopCraft()
    crafting = false
    craftStatus.Text = "已停止"
    craftStatus.TextColor3 = Color3.fromRGB(140, 150, 170)
end

-- ========== 自动合成卖 ==========
local function autoCraftSellOnce()
    if not FF then return end

    local wasRunning = running

    if wasRunning then
        running = false
        roundId = roundId + 1
        currentIndex = 0
        setUI("stopped")
    end

    teleportToMarket()
    task.wait(1)

    local recipe = tonumber(recipeBox.Text) or 2
    local count = tonumber(countBox.Text) or 10
    craftStatus.Text = "自动合成中..."
    craftStatus.TextColor3 = Color3.fromRGB(255, 200, 120)
    for i = 1, count do
        if not autoCraftSellOn then break end
        pcall(function()
            FF:InvokeServer("Craft", {Recipe = recipe})
        end)
        craftStatus.Text = string.format("自动合成 %d/%d", i, count)
        task.wait(0.35)
    end

    task.wait(0.5)
    craftStatus.Text = "自动卖鱼中..."
    local items = scanInventory()
    local sold = 0
    for _, item in ipairs(items) do
        if not autoCraftSellOn then break end
        if isSellFish(item.name) then
            pcall(function()
                FF:InvokeServer("SellFish", {
                    Fish = item.name,
                    Weight = item.weight,
                    Amount = 1
                })
            end)
            sold = sold + 1
            task.wait(0.08)
        end
    end

    craftStatus.Text = string.format("自动完成: 合%d 卖%d", count, sold)
    craftStatus.TextColor3 = Color3.fromRGB(120, 220, 150)

    teleportToSpot()
    task.wait(0.8)

    if wasRunning then
        startFishing()
    end
end

task.spawn(function()
    task.wait(AUTO_CRAFT_SELL_START)
    while true do
        if autoCraftSellOn then
            autoCraftSellOnce()
        end
        task.wait(AUTO_CRAFT_SELL_INTERVAL)
    end
end)
